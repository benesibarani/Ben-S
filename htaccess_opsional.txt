#    2. CADANGKAN .htaccess yang ada: klik kanan - Copy - beri nama
#       htaccess_lama.txt
#    3. Buka .htaccess dengan Edit, lalu SALIN SELURUH ISI DI BAWAH GARIS INI
#       dan tempelkan pada BAGIAN PALING BAWAH berkas (jangan hapus isi lama).
#    4. Simpan, lalu uji:
#          https://rts.benedic-s.com/inbox
#          https://rts.benedic-s.com/gsp
#
#  BILA ADA MASALAH: kembalikan seperti semula (timpa .htaccess dengan
#  htaccess_lama.txt). Aturan ini tidak menyentuh database sama sekali.

# ---------- SALIN MULAI BARIS DI BAWAH INI ----------
<IfModule mod_rewrite.c>
    RewriteEngine On

    # Jangan sentuh berkas dan folder yang memang ada.
    RewriteCond %{REQUEST_FILENAME} -f [OR]
    RewriteCond %{REQUEST_FILENAME} -d
    RewriteRule ^ - [L]

    # /inbox -> inbox.php, /gsp -> gsp.php, dan seterusnya.
    # Hanya berlaku bila berkas .php dengan nama itu benar-benar ada.
    RewriteCond %{REQUEST_FILENAME}\.php -f
    RewriteRule ^(.*?)/?$ $1.php [L,QSA]
</IfModule>
